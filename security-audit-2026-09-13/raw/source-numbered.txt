   37: export class SessionStore {
   38:   readonly file: string;
   39:   constructor(
   40:     readonly directory = sessionDirectory(),
   41:     private readonly keys: KeyStore = new Entry('mun-d2l-mcp', BASE_URL),
   42:   ) { this.file = join(directory, 'session.encrypted.json'); }
   43: 
   44:   async save(session: Session): Promise<void> {
   45:     let secret: string | null | undefined;
   46:     try {
   47:       secret = this.keys.getPassword();
   48:       if (!secret) {
   49:         secret = randomBytes(32).toString('base64');
   50:         this.keys.setPassword(secret);
   51:       }
   52:     } catch { throw new AppError('KEYRING_UNAVAILABLE', 'Windows Credential Manager is unavailable. Session was not saved.'); }
   53:     const key = Buffer.from(secret, 'base64');
   54:     if (key.length !== 32) throw new AppError('KEYRING_INVALID', 'Session encryption key is invalid. Run logout, then login.');
   55:     const iv = randomBytes(12);
   56:     const cipher = createCipheriv('aes-256-gcm', key, iv);
   57:     cipher.setAAD(Buffer.from(BASE_URL));
   58:     const data = Buffer.concat([cipher.update(JSON.stringify(sanitizeSession(session)), 'utf8'), cipher.final()]);
   59:     await mkdir(this.directory, { recursive: true });
   60:     const temporary = join(this.directory, `session-${randomBytes(8).toString('hex')}.tmp`);
   61:     try {
   62:       await writeFile(temporary, JSON.stringify({ version: 1, iv: iv.toString('base64'), tag: cipher.getAuthTag().toString('base64'), data: data.toString('base64') }), { flag: 'wx', mode: 0o600 });
   63:       await rename(temporary, this.file);
   64:     } finally { key.fill(0); await rm(temporary, { force: true }); }
   65:   }
   66: 
   67:   async load(): Promise<Session | null> {
   68:     let content: string;
   69:     try { content = await readFile(this.file, 'utf8'); }
   70:     catch (error) {
   71:       if ((error as NodeJS.ErrnoException).code === 'ENOENT') return null;
   72:       throw new AppError('SESSION_UNREADABLE', 'Cannot read the saved session.');
   73:     }
   74:     try {
   75:       const secret = this.keys.getPassword();
   76:       if (!secret) throw new Error('Missing key');
   77:       if (content.length > 2 * 1024 * 1024) throw new Error('Oversized session envelope');
   78:       const saved = envelope.parse(JSON.parse(content));
   79:       const key = Buffer.from(secret, 'base64');
   80:       const iv = Buffer.from(saved.iv, 'base64');
   81:       const tag = Buffer.from(saved.tag, 'base64');
   82:       if (key.length !== 32 || iv.length !== 12 || tag.length !== 16 || saved.data.length > 2 * 1024 * 1024) throw new Error('Invalid envelope');
   83:       const decipher = createDecipheriv('aes-256-gcm', key, iv);
   84:       decipher.setAAD(Buffer.from(BASE_URL));
   85:       decipher.setAuthTag(tag);
   86:       try { return sanitizeSession(JSON.parse(Buffer.concat([decipher.update(Buffer.from(saved.data, 'base64')), decipher.final()]).toString('utf8'))); }
   87:       finally { key.fill(0); }
   88:     } catch { throw new AppError('SESSION_UNREADABLE', 'Cannot decrypt the session. Run logout, then login.'); }
   89:   }
   90: 
   91:   async clear(): Promise<void> {
   92:     await rm(this.file, { force: true });
   93:     try { if (this.keys.getPassword()) this.keys.deletePassword(); }
   94:     catch { throw new AppError('KEYRING_UNAVAILABLE', 'Session file removed; the encryption key could not be removed from Windows Credential Manager.'); }
   95:   }
   96: }
